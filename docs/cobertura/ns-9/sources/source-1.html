 @Composable

         navBackStackEntry?.destination?.route ?: Routes.WELCOME
 
     val user by sessionViewModel.user.collectAsState()
 
     val context = LocalContext.current
 
     val notificationStore = remember(context.applicationContext) {
         DemoNotificationStore(context.applicationContext)
     }
 
     fun loggedIn(value: DemoUser) {
         sessionViewModel.startSession(value)
 
         nav.navigate(Routes.DASHBOARD) {
             popUpTo(Routes.WELCOME) {
                 inclusive = true
             }
             launchSingleTop = true
         }
     }
 
     fun logOut() {
         sessionViewModel.endSession()
 
         nav.navigate(Routes.LOGIN) {
             popUpTo(Routes.DASHBOARD) {
                 inclusive = true
             }
             launchSingleTop = true
         }
     }
 
     Surface(
         modifier = Modifier.fillMaxSize()
     ) {
         NavHost(
             navController = nav,
             startDestination = if (user == null) {
                 Routes.WELCOME
             } else {
                 Routes.DASHBOARD
             }
         ) {
             // 1. Bienvenida
             composable(Routes.WELCOME) {
                 WelcomeScreen(
                     onBegin = {
                         nav.navigate(Routes.LOGIN)
                     },
                     onLogin = {
                         nav.navigate(Routes.LOGIN)
                     },
                     onAbout = {
                         nav.navigate(Routes.ABOUT)
                     }
                 )
             }
 
             // 2. Registro
             composable(Routes.REGISTER) {
                 val vm: RegisterViewModel = viewModel()
 
                 RegisterScreen(
                     vm,
                     onBack = {
                         nav.popBackStack()
                     },
                     onDone = ::loggedIn
                 )
             }
 
             // 3. Inicio de sesión
             composable(Routes.LOGIN) {
                 val vm: LoginViewModel = viewModel()
 
                 LoginScreen(
                     vm,
                     onBack = {
                         nav.popBackStack()
                     },
                     onAuthenticated = ::loggedIn,
                     onRegister = {
                         nav.navigate(Routes.REGISTER)
                     }
                 )
             }
 
             // 4. Inicio
             composable(Routes.DASHBOARD) {
                 val active = user
 
                 if (active != null) {
                     DashboardScreen(
                         user = active,
                         navController = nav,
                         currentRoute = currentRoute,
                         onAbout = {
                             nav.navigate(Routes.ABOUT)
                         },
                         onLogout = ::logOut,
                         onProfile = {
                             nav.navigate(Routes.PROFILE)
                         },
                         onCatalog = {
                             nav.navigate(Routes.CATALOG)
                         },
                         onCart = {
                             nav.navigate(Routes.CART)
                         },
                         onQr = {
                             nav.navigate(Routes.QR)
                         },
                         onMap = {
                             if (active.role == UserRole.CLIENTE) {
                                 nav.navigate(Routes.MAP)
                             }
                         },
                         onOrders = {
                             nav.navigate(Routes.ORDERS)
                         },
                         onAdmin = {
                             nav.navigate(Routes.PRODUCT_MANAGEMENT)
                         },
                         onNotifications = {
                             nav.navigate(Routes.NOTIFICATIONS)
                         },
                         onPromotions = {
                             nav.navigate(Routes.PROMOTIONS)
                         }
                     )
                 } else {
                     LaunchedEffect(Unit) {
                         nav.navigate(Routes.LOGIN) {
                             popUpTo(Routes.DASHBOARD) {
                                 inclusive = true
                             }
                             launchSingleTop = true
                         }
                     }
                 }
             }
 
             // 5. Mapa: solo cliente
             composable(Routes.MAP) {
                 val active = user
 
                 if (active?.role == UserRole.CLIENTE) {
                     MapScreen(
                         navController = nav,
                         currentRoute = currentRoute,
                         onBack = {
                             nav.popBackStack()
                         }
                     )
                 } else {
                     LaunchedEffect(active?.alias, active?.role) {
                         nav.navigate(
                             if (active == null) {
                                 Routes.LOGIN
                             } else {
                                 Routes.DASHBOARD
                             }
                         ) {
                             popUpTo(Routes.MAP) {
                                 inclusive = true
                             }
                             launchSingleTop = true
                         }
                     }
                 }
             }
 
             // 6. Catálogo
             composable(Routes.CATALOG) {
                 val active = user
 
                 if (active != null) {
                     CatalogScreen(
                         ownerAlias = active.alias,
                         isAdmin = active.role == UserRole.ADMINISTRADOR,
                         navController = nav,
                         currentRoute = currentRoute,
                         onBack = {
                             nav.popBackStack()
                         },
                         onCart = {
                             nav.navigate(Routes.CART)
                         },
                         onProductClick = { productId ->
                             nav.navigate(
                                 Routes.productDetail(productId)
                             )
                         }
                     )
                 }
             }
 
             // 7. Detalle del producto
             composable(
                 route = Routes.PRODUCT_DETAIL,
                 arguments = listOf(
                     navArgument("productId") {
                         type = NavType.StringType
                     }
                 )
             ) { backStackEntry ->
                 val active = user
 
                 val productId =
                     backStackEntry.arguments
                         ?.getString("productId") ?: "OY001"
 
                 if (active != null) {
                     ProductDetailScreen(
                         productId = productId,
                         ownerAlias = active.alias,
                         onBack = {
                             nav.popBackStack()
                         },
                         onCart = {
                             nav.navigate(Routes.CART)
                         },
                         isAdmin = active.role == UserRole.ADMINISTRADOR,
                         onAdjustInventory = {
                             nav.navigate(
                                 "inventory_adjust/$productId"
                             )
                         }
                     )
                 }
             }
 
             // 8. Escáner QR
             composable(Routes.QR) {
                 val active = user
 
                 if (active != null) {
                     QrScannerScreen(
                         isAdmin = active.role == UserRole.ADMINISTRADOR,
                         navController = nav,
                         currentRoute = currentRoute,
                         onProductFound = { productId ->
                             nav.navigate(
                                 Routes.productDetail(productId)
                             )
                         },
                         onAdminInventory = { productId ->
                             nav.navigate(
                                 "inventory_adjust/$productId"
                             )
                         }
                     )
                 }
             }
 
             // 9. Carrito: solo cliente
             composable(Routes.CART) {
                 val active = user
 
                 if (active?.role == UserRole.CLIENTE) {
                     CartScreen(
                         ownerAlias = active.alias,
                         onBack = {
                             nav.popBackStack()
                         },
                         onCheckout = { orderId, _ ->
                             nav.navigate(
                                 Routes.orderConfirmation(orderId)
                             ) {
                                 popUpTo(Routes.CART) {
                                     inclusive = true
                                 }
                             }
                         }
                     )
                 }
             }
 
             // 10. Confirmación del pedido
             composable(
                 route = Routes.ORDER_CONFIRMATION,
                 arguments = listOf(
                     navArgument("orderId") {
                         type = NavType.StringType
                     }
                 )
             ) { backStackEntry ->
                 val active = user
 
                 val orderId =
                     backStackEntry.arguments
                         ?.getString("orderId")
                         .orEmpty()
 
                 if (active?.role == UserRole.CLIENTE) {
                     OrderConfirmationScreen(
                         orderId = orderId,
                         ownerAlias = active.alias,
                         onViewOrders = {
                             nav.navigate(Routes.ORDERS) {
                                 popUpTo(Routes.DASHBOARD)
                                 launchSingleTop = true
                             }
                         },
                         onContinueShopping = {
                             nav.navigate(Routes.CATALOG) {
                                 popUpTo(Routes.DASHBOARD)
                                 launchSingleTop = true
                             }
                         }
                     )
                 } else {
                     LaunchedEffect(active?.alias, active?.role) {
                         nav.navigate(
                             if (active == null) {
                                 Routes.LOGIN
                             } else {
                                 Routes.DASHBOARD
                             }
                         ) {
                             popUpTo(Routes.ORDER_CONFIRMATION) {
                                 inclusive = true
                             }
                             launchSingleTop = true
                         }
                     }
                 }
             }
 
             // 11. Gestión de productos: solo administrador
             composable(Routes.PRODUCT_MANAGEMENT) {
                 val active = user
 
                 if (active?.role == UserRole.ADMINISTRADOR) {
                     ProductManagementScreen(
                         navController = nav,
                         currentRoute = currentRoute
                     )
                 }
             }
 
             // 12. Gestión de inventario: solo administrador
             composable(Routes.INVENTORY_MANAGEMENT) {
                 val active = user
 
                 if (active?.role == UserRole.ADMINISTRADOR) {
                     InventoryScreen(
                         navController = nav,
                         currentRoute = currentRoute
                     )
                 }
             }
 
             // 12b. Ajuste de inventario desde el QR
             composable(
                 route = "inventory_adjust/{productId}",
                 arguments = listOf(
                     navArgument("productId") {
                         type = NavType.StringType
                     }
                 )
             ) { backStackEntry ->
                 val active = user
 
                 val productId =
                     backStackEntry.arguments
                         ?.getString("productId")
                         .orEmpty()
 
                 if (active?.role == UserRole.ADMINISTRADOR) {
                     InventoryScreen(
                         navController = nav,
                         currentRoute = currentRoute,
                         initialQuery = productId
                     )
                 }
             }
 
             // 13. Historial de pedidos
             composable(Routes.ORDERS) {
                 val active = user
 
                 if (active != null) {
                     OrdersScreen(
                         alias = active.alias,
                         isAdmin = active.role == UserRole.ADMINISTRADOR,
                         navController = nav,
                         currentRoute = currentRoute,
                         onBack = {
                             nav.popBackStack()
                         }
                     )
                 }
             }
 
             // 14. Historial de notificaciones
             composable(Routes.NOTIFICATIONS) {
                 val active = user
 
                 if (active != null) {
                     NotificationsScreen(
                         alias = active.alias,
                         onBack = {
                             nav.popBackStack()
                         },
                         onOpen = { target ->
                             when (target) {
                                 "orders" -> {
                                     nav.navigate(Routes.ORDERS) {
                                         launchSingleTop = true
                                     }
                                 }
 
                                 "inventory" -> {
                                     if (
                                         active.role ==
                                         UserRole.ADMINISTRADOR
                                     ) {
                                         nav.navigate(
                                             Routes.INVENTORY_MANAGEMENT
                                         ) {
                                             launchSingleTop = true
                                         }
                                     }
                                 }
                             }
                         }
                     )
                 }
             }
 
             // Promociones: cliente y administrador
             composable(Routes.PROMOTIONS) {
                 val active = user
 
                 if (active != null) {
                     PromotionsScreen(
                         isAdmin = active.role == UserRole.ADMINISTRADOR,
                         onBack = {
                             nav.popBackStack()
                         },
                         onProductClick = { productId ->
                             nav.navigate(
                                 Routes.productDetail(productId)
                             )
                         }
                     )
                 }
             }
 
             // 15. Perfil
             composable(Routes.PROFILE) {
                 val active = user
 
                 if (active != null) {
                     ProfileScreen(
                         user = active,
                         navController = nav,
                         currentRoute = currentRoute,
                         onBack = {
                             nav.popBackStack()
                         },
                         onLogout = ::logOut
                     )
                 } else {
                     LaunchedEffect(Unit) {
                         nav.navigate(Routes.LOGIN) {
                             launchSingleTop = true
                         }
                     }
                 }
             }
 
             // 16. Acerca de
             composable(Routes.ABOUT) {
                 AboutScreen(
                     onBack = {
                         nav.popBackStack()
                     }
                 )
             }
         }
     }
 
     // Apertura desde una notificación del celular.
     LaunchedEffect(
         notificationIntent,
         user?.alias,
         currentRoute
     ) {
         val pending = notificationIntent
         val active = user
 
         if (
             pending != null &&
             active != null &&
             navBackStackEntry != null
         ) {
             val ownerAlias = pending.getStringExtra(
                 "notification_alias"
             )
 
             val target = pending.getStringExtra(
                 "notification_target"
             )
 
             val noticeId = pending.getStringExtra(
                 "notification_id"
             )
 
             // Solo abre avisos del usuario conectado.
             if (ownerAlias == active.alias) {
                 val destination = when (target) {
                     "orders" -> Routes.ORDERS
 
                     "inventory" -> {
                         if (active.role == UserRole.ADMINISTRADOR) {
                             Routes.INVENTORY_MANAGEMENT
                         } else {
                             null
                         }
                     }
 
                     else -> null
                 }
 
                 if (destination != null) {
                     nav.navigate(destination) {
                         launchSingleTop = true
                     }
 
                     if (noticeId != null) {
                         notificationStore.markRead(
                             active.alias,
                             noticeId
                         )
                     }
                 }
             }
 
             onNotificationConsumed()
         }
     }
