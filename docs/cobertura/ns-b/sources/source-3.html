     val context = LocalContext.current
     val app = context.applicationContext as OylApplication
     val repo = app.catalogRepository
 
     val products by repo.products.collectAsState(initial = emptyList())
 
     val cartFlow = remember(repo, ownerAlias) {
         repo.cart(ownerAlias)
     }
 
     val cartItems by key(ownerAlias) {
         cartFlow.collectAsState(initial = emptyList())
     }
 
     val cartCount = cartItems
         .filter { it.quantity > 0 }
         .distinctBy { it.productId }
         .size
 
     var query by remember { mutableStateOf("") }
     var category by remember { mutableStateOf("Todos") }
 
     val scope = rememberCoroutineScope()
 
     var cartMessage by remember(ownerAlias) {
         mutableStateOf<String?>(null)
     }
     var messageIsError by remember(ownerAlias) {
         mutableStateOf(false)
     }
     var messageVersion by remember(ownerAlias) {
         mutableIntStateOf(0)
     }
     var addingToCart by remember(ownerAlias) {
         mutableStateOf(false)
     }
 
     LaunchedEffect(ownerAlias, messageVersion) {

             delay(2500)
             cartMessage = null
         }

             category == "Todos" ||
                     product.category.equals(category, ignoreCase = true)
 
         val matchesQuery =
             product.name.contains(query, ignoreCase = true) ||
                     product.category.contains(query, ignoreCase = true)
 
         matchesCategory && matchesQuery
     }
 
     Scaffold(
         topBar = {
             TopAppBar(
                 title = {
                     Text(
                         text = "Catálogo de Productos",
                         fontWeight = FontWeight.Bold
                     )
                 },
                 actions = {
                     if (!isAdmin) {
                         IconButton(onClick = onCart) {
                             BadgedBox(
                                 badge = {
                                     if (cartCount > 0) {
                                         Badge(
                                             containerColor = Color.Red,
                                             contentColor = Color.White
                                         ) {
                                             Text(
                                                 text = if (cartCount > 99) {
                                                     "99+"
                                                 } else {
                                                     cartCount.toString()
                                                 },
                                                 fontWeight = FontWeight.Bold
                                             )
                                         }
                                     }
                                 }
                             ) {
                                 Icon(
                                     imageVector = Icons.Default.ShoppingCart,
                                     contentDescription = "Carrito: $cartCount productos"
                                 )
                             }
                         }
                     }
                 }
             )
         },
         bottomBar = {
             OylBottomBar(
                 navController = navController,
                 currentRoute = currentRoute,
                 isAdmin = isAdmin
             )
         }
     ) { padding ->
         Column(
             modifier = Modifier
                 .fillMaxSize()
                 .padding(padding)
                 .padding(horizontal = 16.dp),
             verticalArrangement = Arrangement.spacedBy(12.dp)
         ) {
             AnimatedContent(
                 targetState = cartMessage to messageIsError,
                 modifier = Modifier.fillMaxWidth(),
                 transitionSpec = {
                     (
                             slideInVertically(
                                 animationSpec = tween(250),
                                 initialOffsetY = { -it / 2 }
                             ) + fadeIn(tween(250))
                             ) togetherWith (
                             slideOutVertically(
                                 animationSpec = tween(180),
                                 targetOffsetY = { -it / 2 }
                             ) + fadeOut(tween(180))
                             )
                 },
                 label = "Aviso animado del carrito"
             ) { notice ->
                 val message = notice.first
                 val isError = notice.second
 
                 if (message != null) {
                     Surface(
                         modifier = Modifier
                             .fillMaxWidth()
                             .semantics {
                                 liveRegion = LiveRegionMode.Polite
                             },
                         shape = RoundedCornerShape(12.dp),
                         color = if (isError) {
                             MaterialTheme.colorScheme.errorContainer
                         } else {
                             MaterialTheme.colorScheme.primaryContainer
                         }
                     ) {
                         Text(
                             text = message,
                             modifier = Modifier.padding(14.dp),
                             fontWeight = FontWeight.SemiBold,
                             color = if (isError) {
                                 MaterialTheme.colorScheme.onErrorContainer
                             } else {
                                 MaterialTheme.colorScheme.onPrimaryContainer
                             },
                             style = MaterialTheme.typography.bodyMedium
                         )
                     }
                 }
             }

                 value = query,
                 onValueChange = { query = it },
                 modifier = Modifier.fillMaxWidth(),

                 leadingIcon = {
                     Icon(
                         imageVector = Icons.Default.Search,
                         contentDescription = null
                     )
                 },
                 shape = RoundedCornerShape(16.dp),
                 singleLine = true

             Row(
                 modifier = Modifier
                     .fillMaxWidth()
                     .horizontalScroll(rememberScrollState()),
                 horizontalArrangement = Arrangement.spacedBy(8.dp)
             ) {
                 listOf(
                     "Todos",
                     "Frutos secos",
                     "Semillas",
                     "Deshidratados"
                 ).forEach { value ->
                     FilterChip(
                         selected = category == value,
                         onClick = { category = value },
                         label = { Text(value) }
                     )
                 }
             }
 
             LazyVerticalGrid(
                 columns = GridCells.Fixed(2),
                 modifier = Modifier.weight(1f),
                 horizontalArrangement = Arrangement.spacedBy(12.dp),
                 verticalArrangement = Arrangement.spacedBy(12.dp),
                 contentPadding = PaddingValues(bottom = 12.dp)
             ) {
                 items(
                     items = filteredProducts,
                     key = { it.id }
                 ) { product ->
                     Card(
                         modifier = Modifier
                             .fillMaxWidth()
                             .clickable {
                                 onProductClick(product.id)

                             val fallback = painterResource(

                                 fallback = fallback,
                                 modifier = Modifier
                                     .fillMaxWidth()
                                     .height(100.dp)
                                     .clip(RoundedCornerShape(12.dp))
                                     .background(OylMint),
                                 contentScale = ContentScale.Crop
                             )
 
                             Text(
                                 text = product.name,
                                 fontWeight = FontWeight.Bold,
                                 style = MaterialTheme.typography.titleMedium,
                                 maxLines = 1
                             )
 
                             Row(
                                 modifier = Modifier.fillMaxWidth(),
                                 horizontalArrangement = Arrangement.SpaceBetween,
                                 verticalAlignment = Alignment.CenterVertically
                             ) {
                                 Column {
                                     Text(
                                         text = "$${product.price}",
                                         color = OylGreen,
                                         fontWeight = FontWeight.ExtraBold,
                                         style = MaterialTheme.typography.bodyLarge
                                     )
 
                                     Text(
                                         text = "Stock: ${product.stock} un",
                                         style = MaterialTheme.typography.labelSmall,
                                         color = MaterialTheme.colorScheme.secondary
                                     )
                                 }
 
                                 if (!isAdmin) {
                                     var clickAnimation by remember(
                                         ownerAlias,
                                         product.id
                                     ) {
                                         mutableStateOf(false)
                                     }
 
                                     val interaction = remember(product.id) {
                                         MutableInteractionSource()

                                     val hovered by interaction.collectIsHoveredAsState()
                                     val pressed by interaction.collectIsPressedAsState()
 
                                     val buttonEnabled =

                                     val buttonScale by animateFloatAsState(
                                         targetValue = when {
                                             clickAnimation -> 0.75f
                                             pressed -> 0.82f
                                             hovered && buttonEnabled -> 0.90f
                                             else -> 1f
                                         },
                                         animationSpec = spring(
                                             dampingRatio = 0.65f,
                                             stiffness = 500f
                                         ),
                                         label = "Escala del botón agregar"
                                     )
 
                                     IconButton(
                                         interactionSource = interaction,
                                         enabled = buttonEnabled,
                                         onClick = {
                                             if (!addingToCart) {
                                                 addingToCart = true
                                                 clickAnimation = true
 
                                                 scope.launch {
                                                     delay(180)
                                                     clickAnimation = false
                                                 }
 
                                                 scope.launch {
                                                     try {
                                                         val dao = app.database.catalogDao()
 
                                                         val before = dao.getCartItem(
                                                             ownerAlias,
                                                             product.id
                                                         )?.quantity ?: 0
 
                                                         repo.changeCart(
                                                             ownerAlias,
                                                             product.id,
                                                             1
                                                         )
 
                                                         val after = dao.getCartItem(
                                                             ownerAlias,
                                                             product.id
                                                         )?.quantity ?: 0
 
                                                         messageIsError = after <= before
 
                                                         cartMessage = if (after > before) {
                                                             "Añadido: ${product.name}"
                                                         } else {
                                                             "No puedes agregar más unidades de ${product.name}."
                                                         }
 
                                                         messageVersion++
                                                     } catch (e: CancellationException) {
                                                         throw e
                                                     } catch (_: Exception) {

                                                             "No se pudo añadir ${product.name}."
                                                         messageVersion++
                                                     } finally {
                                                         addingToCart = false
                                                     }
                                                 }
                                             }
                                         },
                                         modifier = Modifier
                                             .size(36.dp)
                                             .hoverable(
                                                 interactionSource = interaction,
                                                 enabled = buttonEnabled
                                             )
                                             .graphicsLayer {
                                                 scaleX = buttonScale
                                                 scaleY = buttonScale
                                             }
                                             .clip(CircleShape)
                                             .background(
                                                 if (product.stock > 0) {
                                                     OylGreen
                                                 } else {
                                                     MaterialTheme.colorScheme.outline
                                                 }

                                             imageVector = Icons.Default.Add,

 fun productImage(productId: String): Int {
     return when (productId) {
         "OY001" -> R.drawable.almendras
         "OY002" -> R.drawable.nueces
         "OY003" -> R.drawable.mix_premium
         "OY004" -> R.drawable.pasas_morenas
         "OY005" -> R.drawable.chia
         "OY006" -> R.drawable.cranberries
         else -> R.drawable.frutos_imagen
     }
